**关键修复：**
```javascript
// 对于 /assets/ 路径（前端静态资源）
if (url.startsWith('/assets/')) {
  const isProduction = process.env.NODE_ENV === 'production';
  
  // 生产环境：使用前端服务 URL
  if (isProduction) {
    const frontendUrl = process.env.FRONTEND_URL;
    if (frontendUrl) {
      const baseUrl = frontendUrl.replace(/\/$/, '');
      return `${baseUrl}${url}`;
    }
  }
  
  // 本地开发：保持相对路径，让 Next.js 自动处理
  return url;
}
```
